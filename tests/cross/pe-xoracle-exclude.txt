# pe/x-oracle known-divergence list (W1). One tests/-relative source path per line.
# These run-mode goldens do NOT agree between mcc-PE and the vendored mingw-w64 gcc
# on a native x86_64 Windows host, measured 2026-08-13. Each is an
# implementation-defined / reference-disagreement class, not a confirmed mcc
# miscompile; per-program triage is W2. The oracle counts these but does not fail
# on them. A program that starts agreeing can be removed; a NEW disagreement outside
# this list fails the cell.

# long double: 80-bit (mcc) vs 64-bit (Win64 mingw) representation and conversions
exec/types/floating_point.c
exec/types/floating_point_literals.c
exec/types/ldouble_to_signed.c

# _Complex: mcc's complex support / conversions differ from mingw
exec/features_c99_c11/c11_complex_convert.c

# MS struct bitfield layout edges
exec/structs_unions/bitfields_ms.c
exec/structs_unions/bitfield_width64.c

# Extended-type bit-field integer promotion (T-win-50015): once PE defaults to
# ms_bitfields, mcc promotes an `unsigned long long : 31` bit-field to UNSIGNED
# (its base type's signedness), matching cl.exe / MSVC; mingw/gcc promote it to
# signed via the width-based GCC rule. The standard only fixes promotion for
# int/unsigned-int bit-fields, so extended-type bit-fields are impl-defined and
# MSVC != GCC here — mcc conforms to the native Windows compiler, so this is a
# reference disagreement, not an mcc miscompile (same class as the f80/complex
# rows above). Verified: cl.exe agrees with mcc.
exec/expressions/integer_promotion.c

# __cleanup__ attribute: mcc ignores it (warns), mingw runs the cleanup
exec/features_c99_c11/cleanup.c

# <stdatomic.h> pointer atomics: sequencing/result differs
exec/features_c99_c11/atomic_ptr.c

# freestanding header availability differs between the two toolchains' env
exec/features_c99_c11/c11_freestanding_headers.c

# predefined data-model macros (__LP64__/__LLP64__ etc.) differ
exec/codegen/memory_model.c

# fenv access / FP rounding fold: result depends on FP environment, not stable cross-vendor
exec/features_c99_c11/fenv_access_fold.c

# arch-specific golden (arm64) whose output is not target-neutral on x86_64
exec/arch/arm64.c

# mcc does not compile these on this target (triage pending)
exec/features_c99_c11/versym.c
exec/features_c99_c11/complex_cmplx_special.c
exec/features_c99_c11/tgmath_dispatch.c
